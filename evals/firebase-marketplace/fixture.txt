FICTIONAL TUTORING MARKETPLACE (FLUTTER + FIREBASE) — RULES AND PSEUDOCODE, NOT EXECUTABLE

Product contract:
- Tutor profiles (name, subjects, hourly rate, rating) are public.
- A user's payout details and uploaded ID documents are visible only to that user and admins.
- Everyone signs up as a student. Becoming a tutor requires admin verification, which sets role 'tutor' and verified true.
- The student pays when booking. The tutor is paid when the student marks the session completed.
- Chat messages are visible only to the two participants.
- Only the booking's student or tutor can cancel it, and only before it is completed; cancelling refunds the student.
- Admins have the custom claim admin == true.

firestore.rules:
  rules_version = '2';
  service cloud.firestore {
    match /databases/{database}/documents {
      function signedIn() { return request.auth != null; }
      function isAdmin() { return signedIn() && request.auth.token.admin == true; }
      function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }

      match /tutors/{tutorId} {
        allow read: if true;
        allow write: if signedIn() && request.auth.uid == tutorId && me().role == 'tutor' && me().verified == true;
      }

      match /users/{uid} {
        allow create: if signedIn() && request.auth.uid == uid && request.resource.data.role == 'student' && request.resource.data.verified == false;
        allow update: if signedIn() && request.auth.uid == uid
          && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['rating', 'ratingCount']);
        allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      }
      match /users/{uid}/{document=**} {
        allow read: if signedIn();
      }
      match /users/{uid}/private/{doc} {
        allow read, write: if signedIn() && request.auth.uid == uid;
      }

      match /bookings/{bookingId} {
        allow create: if signedIn() && request.resource.data.studentId == request.auth.uid
          && request.resource.data.status == 'pending_payment' && !('priceCents' in request.resource.data);
        allow read: if signedIn() && request.auth.uid in [resource.data.studentId, resource.data.tutorId];
        allow update: if signedIn() && request.auth.uid in [resource.data.studentId, resource.data.tutorId]
          && request.resource.data.studentId == resource.data.studentId
          && request.resource.data.tutorId == resource.data.tutorId;
      }

      match /chats/{chatId} {
        allow read: if signedIn() && request.auth.uid in resource.data.participants;
        allow create: if signedIn() && request.resource.data.participants.size() == 2 && request.auth.uid in request.resource.data.participants;
        match /messages/{messageId} {
          allow read: if signedIn() && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
          allow create: if signedIn() && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants
            && request.resource.data.senderId == request.auth.uid;
        }
      }

      match /reviews/{reviewId} {
        allow read: if true;
        allow create: if signedIn() && request.resource.data.studentId == request.auth.uid
          && get(/databases/$(database)/documents/bookings/$(request.resource.data.bookingId)).data.studentId == request.auth.uid
          && get(/databases/$(database)/documents/bookings/$(request.resource.data.bookingId)).data.status == 'completed';
      }
    }
  }

storage.rules:
  rules_version = '2';
  service firebase.storage {
    match /b/{bucket}/o {
      match /avatars/{uid}/{file} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == uid
          && request.resource.size < 2 * 1024 * 1024 && request.resource.contentType.matches('image/.*');
      }
      match /id-documents/{uid}/{file} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == uid;
      }
    }
  }

functions/index.js (pseudocode, Admin SDK):
  exports.createPaymentIntent = onCall(async (request) => {
    if (!request.auth) throw new HttpsError('unauthenticated')
    booking = await db.doc(`bookings/${request.data.bookingId}`).get()
    if (booking.studentId != request.auth.uid) throw new HttpsError('permission-denied')
    tutor = await db.doc(`tutors/${booking.tutorId}`).get()
    amount = tutor.hourlyRateCents * booking.hours
    pi = await stripe.paymentIntents.create({ amount, currency: 'usd', metadata: { bookingId: booking.id } })
    await booking.ref.update({ paymentIntentId: pi.id, priceCents: amount })
    return { clientSecret: pi.client_secret }
  })

  exports.stripeWebhook = onRequest(async (req, res) => {
    event = stripe.webhooks.constructEvent(req.rawBody, req.headers['stripe-signature'], WEBHOOK_SECRET)   // throws on bad signature
    if (event.type == 'payment_intent.succeeded')
      await db.doc(`bookings/${event.data.object.metadata.bookingId}`).update({ status: 'confirmed' })
    res.sendStatus(200)
  })

  exports.onBookingUpdated = onDocumentUpdated('bookings/{id}', async (event) => {
    before = event.data.before.data(); after = event.data.after.data()
    if (before.status != 'completed' && after.status == 'completed') {
      tutor = await db.doc(`users/${after.tutorId}/private/payout`).get()
      await stripe.transfers.create({ amount: Math.round(after.priceCents * 0.85), destination: tutor.stripeAccountId })
    }
  })

  exports.cancelBooking = onCall(async (request) => {
    if (!request.auth) throw new HttpsError('unauthenticated')
    booking = await db.doc(`bookings/${request.data.bookingId}`).get()
    await stripe.refunds.create({ payment_intent: booking.paymentIntentId })
    await booking.ref.update({ status: 'cancelled' })
  })

  exports.verifyTutor = onCall(async (request) => {
    if (request.auth?.token.admin !== true) throw new HttpsError('permission-denied')
    await db.doc(`users/${request.data.uid}`).update({ role: 'tutor', verified: true })
  })
