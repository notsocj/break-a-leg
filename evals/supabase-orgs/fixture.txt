Product contract:
- Users belong to organizations through memberships with role 'owner', 'admin', or 'member'.
- Members can read their organization's clients, invoices, and attachments. Only owners and admins create or edit invoices and invite people.
- People join an organization only by accepting an invitation sent by an owner or admin.
- Plans: free, team, business. The plan comes from what the organization paid for in Stripe.
- A document can be shared read-only through a link that expires at its expires_at time.
- The plan catalog and user avatars are public.

supabase/migrations/20260301000000_orgs.sql:
  create table public.organizations (id uuid primary key default gen_random_uuid(), name text not null, plan text not null default 'free', stripe_customer text);
  create table public.memberships (org_id uuid references public.organizations on delete cascade, user_id uuid references auth.users on delete cascade, role text not null default 'member', primary key (org_id, user_id));
  create table public.invitations (id uuid primary key default gen_random_uuid(), org_id uuid references public.organizations, email text not null, role text not null default 'member', accepted_at timestamptz);
  alter table public.organizations enable row level security;
  alter table public.memberships enable row level security;
  alter table public.invitations enable row level security;

  create function public.is_org_member(p_org uuid) returns boolean
  language sql stable security definer set search_path = '' as $$
    select exists (select 1 from public.memberships where org_id = p_org and user_id = (select auth.uid()));
  $$;
  create function public.is_org_admin(p_org uuid) returns boolean
  language sql stable security definer set search_path = '' as $$
    select exists (select 1 from public.memberships where org_id = p_org and user_id = (select auth.uid()) and role in ('owner', 'admin'));
  $$;

  create policy "members read org" on public.organizations for select to authenticated using (public.is_org_member(id));
  create policy "members read memberships" on public.memberships for select to authenticated using (public.is_org_member(org_id));
  create policy "join org" on public.memberships for insert to authenticated with check (user_id = (select auth.uid()));
  create policy "admins manage invitations" on public.invitations for all to authenticated using (public.is_org_admin(org_id)) with check (public.is_org_admin(org_id));
  create policy "invitee reads invitation" on public.invitations for select to authenticated using (email = (select auth.jwt() ->> 'email'));
  revoke update on public.organizations from authenticated;

supabase/functions/create-checkout/index.ts (pseudocode):
  user = await supabase.auth.getUser(req.headers.authorization)
  { orgId, priceId, plan } = await req.json()
  if (!await isOrgAdmin(user.id, orgId)) return 403
  session = stripe.checkout.sessions.create({ mode: 'subscription', line_items: [{ price: priceId, quantity: 1 }],
    client_reference_id: orgId, metadata: { plan } })
  return { url: session.url }
